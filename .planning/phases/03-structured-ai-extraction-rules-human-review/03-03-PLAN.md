# Plan 03-03: Company Rules Engine — Deterministic Server-Side Verification Layer

## Goal

Build the deterministic server-side rules engine that verifies and normalizes AI-extracted content against v1 company formatting rules. This is the second layer of the dual-layer approach:

1. **Layer 1 (03-01):** AI prompt hardcodes rules, extracts facts accordingly
2. **Layer 2 (this plan):** Server verifies/fixes business-critical fields deterministically AFTER AI extraction

The engine is a pure function `applyRules(draft)` that returns a corrected draft plus a list of violations. Auto-fixable issues (greeting text, return wording) are corrected silently. Non-fixable issues are flagged for human review in 03-02's UI.

### Architecture
```
src/lib/rules/
  types.ts          # Rule, RuleViolation, RuleResult interfaces
  definitions.ts    # 7 v1 rule definitions as pure checker functions
  engine.ts         # applyRules(draft) orchestrator
  seed.ts           # Database seeder for CompanyRule records
  __tests__/
    engine.test.ts  # Unit tests for all 7 rules
```

### Integration Flow
```
extractTour(text)
  -> StructuredDraft (schema-validated by Zod)
  -> applyRules(draft)
  -> { correctedDraft, violations[] }
  -> saveDraft(uploadId, correctedDraft, ..., violations)
  -> Persisted to DB with reviewFlags
```

       import type { StructuredDraft } from "@/lib/ai/extraction-schema";

       /**
        * Severity of a rule violation:
        * - "auto_fixed": The engine corrected the value automatically (high confidence fix)
        * - "needs_review": The engine detected an issue but cannot auto-fix; flagged for human review
        */
       export type ViolationSeverity = "auto_fixed" | "needs_review";

       /**
        * A single rule violation found during deterministic verification.
        */
       export interface RuleViolation {
         /** Rule identifier, e.g., "RULE-03" */
         ruleId: string;
         /** The draft field path that was checked, e.g., "greetingText" */
         field: string;
         /** Human-readable Vietnamese description of the violation */
         message: string;
         /** Whether the engine auto-fixed the issue or flagged it for review */
         severity: ViolationSeverity;
         /** The original value before auto-fix (null if field was missing) */
         originalValue: string | null;
         /** The corrected value after auto-fix (null if severity is "needs_review") */
         correctedValue: string | null;
       }

       /**
        * Result of running the rules engine against a draft.
        */
       export interface RuleResult {
         /** The draft with auto-fixes applied */
         correctedDraft: StructuredDraft;
         /** All violations found (both auto-fixed and needs-review) */
         violations: RuleViolation[];
         /** Count of violations that were auto-fixed */
         autoFixCount: number;
         /** Count of violations that need human review */
         needsReviewCount: number;
       }

       /**
        * A single rule definition — a pure function that checks one aspect of the draft.
        * Returns the (possibly modified) draft and any violations found.
        */
       export interface RuleDefinition {
         /** Rule identifier, e.g., "RULE-03" */
         ruleId: string;
         /** Human-readable name */
         name: string;
         /** The check function: receives a draft, returns modified draft + violations */
         check: (draft: StructuredDraft) => {
           draft: StructuredDraft;
           violations: RuleViolation[];
         };
       }
       ```

    import type { StructuredDraft } from "@/lib/ai/extraction-schema";
    import type { RuleResult, RuleViolation } from "./types";
    import { V1_RULES } from "./definitions";

    /**
     * Run all active v1 rules against a structured draft.
     *
     * This is the second layer of the dual-layer approach:
     * 1. AI prompt hardcodes rules (first layer — 03-01)
     * 2. This engine verifies/fixes deterministically (second layer — this function)
     *
     * Rules are executed in order. Each rule receives the draft (possibly modified
     * by previous rules) and returns the updated draft + any violations.
     *
     * Auto-fixable violations are applied to the draft silently.
     * Non-fixable violations are collected for human review.
     *
     * @param draft - The schema-validated StructuredDraft from AI extraction
     * @returns RuleResult with correctedDraft, violations[], and counts
     */
    export function applyRules(draft: StructuredDraft): RuleResult {
      let currentDraft = structuredClone(draft);
      const allViolations: RuleViolation[] = [];

      for (const rule of V1_RULES) {
        const result = rule.check(currentDraft);
        currentDraft = result.draft;
        allViolations.push(...result.violations);
      }

      const autoFixCount = allViolations.filter(
        (v) => v.severity === "auto_fixed"
      ).length;
      const needsReviewCount = allViolations.filter(
        (v) => v.severity === "needs_review"
      ).length;

      return {
        correctedDraft: currentDraft,
        violations: allViolations,
        autoFixCount,
        needsReviewCount,
      };
    }

    /**
     * Convert rule violations into reviewFlag strings for persistence.
     * These are merged into the Upload.reviewFlags JSON array.
     *
     * Format: "rule:{ruleId}:{severity}" e.g., "rule:RULE-03:auto_fixed"
     */
    export function violationsToReviewFlags(violations: RuleViolation[]): string[] {
      return violations.map(
        (v) => `rule:${v.ruleId}:${v.severity}`
      );
    }
    ```
