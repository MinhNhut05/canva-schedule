# Plan 04-02: Canva API Adapter, OAuth Token Lifecycle & Async Job Orchestration

## Objective

Implement the complete server-side Canva integration layer: OAuth token persistence and auto-refresh, an authenticated REST client with retry-on-401, async autofill job creation and bounded polling, design URL resolution, and the top-level adapter that the review page Server Actions will call. After this plan, the backend can create editable Canva designs from autofill payloads and return fresh edit URLs for persisted design IDs.

**API path note:** This plan implements the official Canva Connect Autofill API path (Brand Templates with datasets). The adapter boundary means if the production account proves to have different capabilities (regular designs on Pro), only the adapter internals change — the adapter contract, persistence, and UI layers remain stable. The CONTEXT.md locked decision D-07 (copy + populate) is respected at the adapter contract level: the template stays untouched, a new design is created from it with populated content.

**Create `src/lib/canva/oauth.ts`:**

```typescript
import "server-only";
import { db } from "@/lib/db";
import { getCanvaConfig } from "./server-client";

const CANVA_TOKEN_URL = "https://api.canva.com/rest/v1/oauth/token";

interface TokenResponse {
  access_token: string;
  refresh_token: string;
  expires_in: number; // seconds
  token_type: string;
  scope?: string;
}

/**
 * Get the current valid access token.
 * 1. Check DB for a non-expired token
 * 2. If expired or missing, refresh using stored or env refresh token
 * 3. Persist the new token set atomically
 */
export async function getValidAccessToken(): Promise<string> {
  const stored = await db.canvaToken.findFirst({ orderBy: { updatedAt: "desc" } });

  if (stored && stored.expiresAt > new Date(Date.now() + 60_000)) {
    return stored.accessToken;
  }

  const refreshToken = stored?.refreshToken ?? getCanvaConfig().refreshToken;
  const newTokens = await refreshAccessToken(refreshToken);

  await db.canvaToken.upsert({
    where: { id: stored?.id ?? "seed" },
    create: {
      accessToken: newTokens.access_token,
      refreshToken: newTokens.refresh_token,
      expiresAt: new Date(Date.now() + newTokens.expires_in * 1000),
      tokenType: newTokens.token_type,
      scope: newTokens.scope ?? null,
    },
    update: {
      accessToken: newTokens.access_token,
      refreshToken: newTokens.refresh_token,
      expiresAt: new Date(Date.now() + newTokens.expires_in * 1000),
      tokenType: newTokens.token_type,
      scope: newTokens.scope ?? null,
    },
  });

  return newTokens.access_token;
}

async function refreshAccessToken(refreshToken: string): Promise<TokenResponse> {
  const config = getCanvaConfig();
  const params = new URLSearchParams({
    grant_type: "refresh_token",
    refresh_token: refreshToken,
  });

  const response = await fetch(CANVA_TOKEN_URL, {
    method: "POST",
    headers: {
      Authorization: `Basic ${Buffer.from(`${config.clientId}:${config.clientSecret}`).toString("base64")}`,
      "Content-Type": "application/x-www-form-urlencoded",
    },
    body: params.toString(),
    cache: "no-store",
  });

  if (!response.ok) {
    const errorText = await response.text().catch(() => "unknown");
    throw new Error(`Canva token refresh failed (${response.status}): ${errorText}`);
  }

  return response.json() as Promise<TokenResponse>;
}
```

**Create `src/lib/canva/client.ts`:**

Authenticated REST wrapper with auto-refresh retry:
```typescript
import "server-only";
import { getValidAccessToken } from "./oauth";

const CANVA_API_BASE = "https://api.canva.com/rest/v1";

export async function canvaFetch(path: string, init?: RequestInit & { retried?: boolean }): Promise<Response> {
  const token = await getValidAccessToken();
  const url = path.startsWith("http") ? path : `${CANVA_API_BASE}${path}`;

  const response = await fetch(url, {
    ...init,
    headers: {
      ...init?.headers,
      Authorization: `Bearer ${token}`,
      "Content-Type": "application/json",
    },
    cache: "no-store",
  });

  // Retry once on 401 (token may have just expired)
  if (response.status === 401 && !init?.retried) {
    return canvaFetch(path, { ...init, retried: true });
  }

  return response;
}
```

**Create `src/lib/canva/jobs.ts`:**

Async autofill job creation + bounded polling:
```typescript
import "server-only";
import { canvaFetch } from "./client";

export interface AutofillJobResult {
  designId: string;
  editUrl: string;
  viewUrl: string;
  thumbnailUrl?: string;
}

export async function createAutofillJob(
  templateId: string,
  data: Record<string, { type: "text"; text: string }>,
  title: string
): Promise<string> {
  const response = await canvaFetch("/autofills", {
    method: "POST",
    body: JSON.stringify({
      brand_template_id: templateId,
      title,
      data,
    }),
  });

  if (!response.ok) {
    const errorText = await response.text().catch(() => "unknown");
    throw new Error(`Canva autofill creation failed (${response.status}): ${errorText}`);
  }

  const payload = await response.json();
  return payload.job.id as string;
}

const POLL_DELAYS_MS = [2000, 3000, 5000, 5000, 5000, 5000];

export async function pollAutofillJob(jobId: string): Promise<AutofillJobResult> {
  for (const delayMs of POLL_DELAYS_MS) {
    await new Promise((resolve) => setTimeout(resolve, delayMs));

    const response = await canvaFetch(`/autofills/${jobId}`);
    if (!response.ok) throw new Error(`Canva poll failed (${response.status})`);

    const payload = await response.json();
    const job = payload.job;

    if (job.status === "success") {
      const design = job.result.design;
      return {
        designId: design.id,
        editUrl: design.urls?.edit_url ?? "",
        viewUrl: design.urls?.view_url ?? "",
        thumbnailUrl: design.thumbnail?.url,
      };
    }

    if (job.status === "failed") {
      throw new Error(job.error?.message ?? "Canva autofill failed");
    }
    // status === "in_progress" -> continue polling
  }

  throw new Error("Canva generation timed out after polling");
}
```

Top-level server-side adapter that orchestrates the full generation flow:

```typescript
import "server-only";
import { db } from "@/lib/db";
import { resolveTemplateId, type TourDuration, type ArtifactKind } from "./template-resolver";
import { createAutofillJob, pollAutofillJob } from "./jobs";
import { getFreshDesignUrls, type DesignUrls } from "./designs";

export interface GenerateArtifactInput {
  uploadId: string;
  duration: TourDuration;
  kind: ArtifactKind;
  data: Record<string, { type: "text"; text: string }>;
  title: string;
}

export interface ArtifactResult {
  artifactType: ArtifactKind;
  status: "SUCCEEDED" | "FAILED";
  designId?: string;
  editUrl?: string;
  viewUrl?: string;
  thumbnailUrl?: string;
  errorMessage?: string;
}

/**
 * Generate a single Canva artifact: resolve template → create job → poll → persist result.
 * The template stays untouched — a new design is created from it with populated content.
 */
export async function generateArtifact(input: GenerateArtifactInput): Promise<ArtifactResult> {
  const templateId = resolveTemplateId(input.duration, input.kind);

  // Upsert artifact record to PROCESSING
  await db.canvaArtifact.upsert({
    where: { uploadId_artifactType: { uploadId: input.uploadId, artifactType: input.kind } },
    create: { uploadId: input.uploadId, artifactType: input.kind, status: "PROCESSING", templateId },
    update: { status: "PROCESSING", templateId, jobId: null, designId: null, errorMessage: null },
  });

  try {
    const jobId = await createAutofillJob(templateId, input.data, input.title);

    await db.canvaArtifact.update({
      where: { uploadId_artifactType: { uploadId: input.uploadId, artifactType: input.kind } },
      data: { jobId },
    });

    const result = await pollAutofillJob(jobId);

    await db.canvaArtifact.update({
      where: { uploadId_artifactType: { uploadId: input.uploadId, artifactType: input.kind } },
      data: { status: "SUCCEEDED", designId: result.designId },
    });

    return {
      artifactType: input.kind,
      status: "SUCCEEDED",
      designId: result.designId,
      editUrl: result.editUrl,
      viewUrl: result.viewUrl,
      thumbnailUrl: result.thumbnailUrl,
    };
  } catch (error) {
    const errorMessage = error instanceof Error ? error.message : "Unknown error";

    await db.canvaArtifact.update({
      where: { uploadId_artifactType: { uploadId: input.uploadId, artifactType: input.kind } },
      data: { status: "FAILED", errorMessage },
    });

    return { artifactType: input.kind, status: "FAILED", errorMessage };
  }
}

/**
 * Resolve fresh URLs for a persisted artifact by its designId.
 */
export async function resolveArtifactUrls(designId: string): Promise<DesignUrls> {
  return getFreshDesignUrls(designId);
}

/**
 * Load persisted artifact state for an upload.
 */
export async function getArtifactsForUpload(uploadId: string) {
  return db.canvaArtifact.findMany({ where: { uploadId }, orderBy: { artifactType: "asc" } });
}
```

This adapter:
- Takes typed input (uploadId, duration, kind, data, title)
- Resolves template from env via template-resolver
- Creates async autofill job, polls to completion
- Persists per-artifact state (PROCESSING → SUCCEEDED/FAILED) with designId
- Returns typed ArtifactResult
- The `generateArtifact` function is called once per artifact — the caller (Plan 04-03 Server Action) calls it twice in parallel via Promise.allSettled for Itinerary and Menu
