# Plan 04-03: Reviewed-Content-to-Editable-Link Flow End to End

## Objective

Wire the complete user-facing flow: after approving content on the review page, user confirms the auto-selected template pair, clicks "Tạo Canva" to generate both Itinerary and Menu in parallel, sees inline progress, and receives two result cards with "Mở trong Canva" and "Sao chép link" actions. Partial success, retry, re-generation, rate limiting, and persistent revisit behavior are all handled.

```typescript
"use server";

import { generateArtifact, getArtifactsForUpload, resolveArtifactUrls, type ArtifactResult } from "@/lib/canva/adapter";
import { resolveTemplatePair } from "@/lib/canva/template-resolver";
import {
  buildOneDayItineraryPayload,
  buildOneDayMenuPayload,
  buildTwoDayItineraryPayload,
  buildTwoDayMenuPayload,
} from "@/lib/canva/payload";
import { getDraft } from "@/lib/review/draft";
import { db } from "@/lib/db";
import type { TourDuration, ArtifactKind } from "@/lib/canva/template-resolver";

export async function generateCanva(uploadId: string): Promise<{
  success: boolean;
  results: ArtifactResult[];
  error?: string;
}> {
  // Load upload + draft
  const upload = await db.upload.findUnique({ where: { id: uploadId } });
  if (!upload || upload.reviewStatus !== "APPROVED") {
    return { success: false, results: [], error: "Nội dung chưa được duyệt." };
  }

  const draft = await getDraft(uploadId);
  if (!draft) {
    return { success: false, results: [], error: "Không tìm thấy bản nháp." };
  }

  const duration = upload.tourDuration as TourDuration;
  if (!duration || !["ONE_DAY", "TWO_DAY"].includes(duration)) {
    return { success: false, results: [], error: "Loại tour không hợp lệ." };
  }

  const pair = resolveTemplatePair(duration);

  // Build payloads based on duration
  const itineraryPayload = duration === "ONE_DAY"
    ? buildOneDayItineraryPayload(draft)
    : buildTwoDayItineraryPayload(draft);
  const menuPayload = duration === "ONE_DAY"
    ? buildOneDayMenuPayload(draft)
    : buildTwoDayMenuPayload(draft);

  // Generate both artifacts in parallel (D-20)
  const [itineraryResult, menuResult] = await Promise.allSettled([
    generateArtifact({
      uploadId,
      duration,
      kind: "ITINERARY",
      data: itineraryPayload,
      title: `SileTravel - ${pair.displayLabel} - Lịch trình`,
    }),
    generateArtifact({
      uploadId,
      duration,
      kind: "MENU",
      data: menuPayload,
      title: `SileTravel - ${pair.displayLabel} - Thực đơn`,
    }),
  ]);

  const results: ArtifactResult[] = [
    itineraryResult.status === "fulfilled"
      ? itineraryResult.value
      : { artifactType: "ITINERARY" as ArtifactKind, status: "FAILED" as const, errorMessage: (itineraryResult.reason as Error)?.message ?? "Lỗi không xác định" },
    menuResult.status === "fulfilled"
      ? menuResult.value
      : { artifactType: "MENU" as ArtifactKind, status: "FAILED" as const, errorMessage: (menuResult.reason as Error)?.message ?? "Lỗi không xác định" },
  ];

  const hasAnySuccess = results.some((r) => r.status === "SUCCEEDED");
  return { success: hasAnySuccess, results };
}

export async function retryCanvaArtifact(uploadId: string, kind: ArtifactKind): Promise<ArtifactResult> {
  // Same as generateCanva but for a single artifact
  const upload = await db.upload.findUnique({ where: { id: uploadId } });
  if (!upload || upload.reviewStatus !== "APPROVED") {
    return { artifactType: kind, status: "FAILED", errorMessage: "Nội dung chưa được duyệt." };
  }

  const draft = await getDraft(uploadId);
  if (!draft) {
    return { artifactType: kind, status: "FAILED", errorMessage: "Không tìm thấy bản nháp." };
  }

  const duration = upload.tourDuration as TourDuration;
  const pair = resolveTemplatePair(duration);

  let payload: Record<string, { type: "text"; text: string }>;
  if (kind === "ITINERARY") {
    payload = duration === "ONE_DAY" ? buildOneDayItineraryPayload(draft) : buildTwoDayItineraryPayload(draft);
  } else {
    payload = duration === "ONE_DAY" ? buildOneDayMenuPayload(draft) : buildTwoDayMenuPayload(draft);
  }

  return generateArtifact({
    uploadId,
    duration,
    kind,
    data: payload,
    title: `SileTravel - ${pair.displayLabel} - ${kind === "ITINERARY" ? "Lịch trình" : "Thực đơn"}`,
  });
}

export async function loadCanvaArtifacts(uploadId: string) {
  const artifacts = await getArtifactsForUpload(uploadId);

  // Resolve fresh URLs for succeeded artifacts
  const withUrls = await Promise.all(
    artifacts.map(async (artifact) => {
      if (artifact.status === "SUCCEEDED" && artifact.designId) {
        try {
          const urls = await resolveArtifactUrls(artifact.designId);
          return { ...artifact, editUrl: urls.editUrl, viewUrl: urls.viewUrl, thumbnailUrl: urls.thumbnailUrl };
        } catch {
          return { ...artifact, editUrl: "", viewUrl: "", thumbnailUrl: undefined };
        }
      }
      return { ...artifact, editUrl: "", viewUrl: "", thumbnailUrl: undefined };
    })
  );

  return withUrls;
}
```
