# Plan 04-03: Reviewed-Content-to-Editable-Link Flow End to End

## Goal

Wire the complete user-facing flow: split approval from generation in the review page, show the auto-selected template confirmation panel, trigger parallel Canva generation via Server Actions calling the adapter, display generation progress inline, show result cards with open/copy/retry actions, handle partial success and rate limiting, and persist + rehydrate Canva links on page revisit. After this plan, a user can approve reviewed content, confirm templates, click "Tao Canva", and receive two editable Canva links directly on `/review/[id]`.

```typescript
export async function generateCanva(
  uploadId: string,
): Promise<{
  success: boolean;
  result?: GeneratePairResult;
  error?: string;
  rateLimitRetryAfter?: number;
}> {
  const userId = await requireAuth();
  if (!userId) {
    return { success: false, error: "Phien dang nhap het han." };
  }

  const upload = await prisma.upload.findFirst({
    where: { id: uploadId, userId },
    select: { id: true, reviewStatus: true, structuredDraft: true },
  });

  if (!upload) {
    return { success: false, error: "Khong tim thay tai lieu." };
  }

  if (upload.reviewStatus !== "APPROVED") {
    return { success: false, error: "Noi dung chua duoc duyet." };
  }

  const draft = await getDraft(uploadId);
  if (!draft) {
    return { success: false, error: "Khong tim thay noi dung da duyet." };
  }

  try {
    const result = await generateCanvaPair(uploadId, draft);

    // Check if either artifact is rate limited
    const rateLimitRetryAfter =
      result.itinerary.rateLimitRetryAfter ?? result.menu.rateLimitRetryAfter;

    revalidatePath(`/review/${uploadId}`);
    return { success: true, result, rateLimitRetryAfter };
  } catch (error) {
    const message =
      error instanceof Error
        ? error.message
        : "Co loi xay ra khi tao Canva.";

    revalidatePath(`/review/${uploadId}`);
    return { success: false, error: message };
  }
}
```

```typescript
export async function retryCanvaArtifact(
  uploadId: string,
  artifactKind: "ITINERARY" | "MENU",
): Promise<{
  success: boolean;
  result?: GenerateArtifactResult;
  error?: string;
  rateLimitRetryAfter?: number;
}> {
  const userId = await requireAuth();
  if (!userId) {
    return { success: false, error: "Phien dang nhap het han." };
  }

  const upload = await prisma.upload.findFirst({
    where: { id: uploadId, userId },
    select: { id: true, reviewStatus: true },
  });

  if (!upload || upload.reviewStatus !== "APPROVED") {
    return { success: false, error: "Khong the thu lai. Noi dung chua duoc duyet." };
  }

  const draft = await getDraft(uploadId);
  if (!draft) {
    return { success: false, error: "Khong tim thay noi dung da duyet." };
  }

  try {
    const result = await retryArtifact(uploadId, artifactKind, draft);
    revalidatePath(`/review/${uploadId}`);
    return { success: true, result, rateLimitRetryAfter: result.rateLimitRetryAfter };
  } catch (error) {
    const message =
      error instanceof Error
        ? error.message
        : "Co loi xay ra khi thu lai.";

    revalidatePath(`/review/${uploadId}`);
    return { success: false, error: message };
  }
}
```

import { notFound, redirect } from "next/navigation";

import { ReviewPage } from "@/components/review/review-page";
import { auth } from "@/lib/auth";
import { prisma } from "@/lib/db";
import { getDraft } from "@/lib/review/draft";
import { getArtifactStates, resolveArtifactLinks } from "@/lib/canva/adapter";

interface ReviewPageRouteProps {
  params: Promise<{ id: string }>;
}

export default async function ReviewPageRoute({
  params,
}: ReviewPageRouteProps) {
  const session = await auth();

  if (!session?.user?.id) {
    redirect("/login");
  }

  const { id } = await params;

  const upload = await prisma.upload.findFirst({
    where: { id, userId: session.user.id },
    select: {
      id: true,
      originalFileName: true,
      aiStatus: true,
      reviewStatus: true,
      aiErrorMessage: true,
      clientType: true,
      tourDuration: true,
    },
  });

  if (!upload) {
    notFound();
  }

  const [draft, artifactStates] = await Promise.all([
    getDraft(id),
    getArtifactStates(id),
  ]);

  // Resolve fresh Canva links if artifacts exist with design IDs
  let canvaLinks: { itinerary: { editUrl: string; viewUrl: string; thumbnailUrl?: string } | null; menu: { editUrl: string; viewUrl: string; thumbnailUrl?: string } | null } = {
    itinerary: null,
    menu: null,
  };

  if (artifactStates.itinerary?.designId || artifactStates.menu?.designId) {
    try {
      const resolved = await resolveArtifactLinks(id);
      canvaLinks = {
        itinerary: resolved.itinerary
          ? { editUrl: resolved.itinerary.editUrl, viewUrl: resolved.itinerary.viewUrl, thumbnailUrl: resolved.itinerary.thumbnailUrl }
          : null,
        menu: resolved.menu
          ? { editUrl: resolved.menu.editUrl, viewUrl: resolved.menu.viewUrl, thumbnailUrl: resolved.menu.thumbnailUrl }
          : null,
      };
    } catch {
      // If link resolution fails, show artifacts with status but no clickable links
    }
  }

  return (
    <ReviewPage
      upload={{
        id: upload.id,
        originalFileName: upload.originalFileName,
        aiStatus: upload.aiStatus,
        reviewStatus: upload.reviewStatus,
        aiErrorMessage: upload.aiErrorMessage,
        clientType: upload.clientType,
        tourDuration: upload.tourDuration,
      }}
      draft={draft}
      canvaState={{
        itinerary: artifactStates.itinerary
          ? {
              status: artifactStates.itinerary.status,
              designId: artifactStates.itinerary.designId,
              errorMessage: artifactStates.itinerary.errorMessage,
              editUrl: canvaLinks.itinerary?.editUrl ?? null,
              viewUrl: canvaLinks.itinerary?.viewUrl ?? null,
              thumbnailUrl: canvaLinks.itinerary?.thumbnailUrl ?? null,
            }
          : null,
        menu: artifactStates.menu
          ? {
              status: artifactStates.menu.status,
              designId: artifactStates.menu.designId,
              errorMessage: artifactStates.menu.errorMessage,
              editUrl: canvaLinks.menu?.editUrl ?? null,
              viewUrl: canvaLinks.menu?.viewUrl ?? null,
              thumbnailUrl: canvaLinks.menu?.thumbnailUrl ?? null,
            }
          : null,
      }}
    />
  );
}
```

"use client";

import { useEffect, useState } from "react";
import { Alert, AlertDescription, AlertTitle } from "@/components/ui/alert";

interface CanvaGenerationPanelProps {
  isGenerating: boolean;
  error?: string | null;
  rateLimitRetryAfter?: number | null;
}

function SpinnerIcon() {
  return (
    <svg
      aria-hidden="true"
      viewBox="0 0 24 24"
      fill="none"
      stroke="currentColor"
      strokeWidth="2"
      strokeLinecap="round"
      strokeLinejoin="round"
      className="size-5 animate-spin text-[#3B82F6]"
    >
      <path d="M21 12a9 9 0 1 1-6.219-8.56" />
    </svg>
  );
}

export function CanvaGenerationPanel({
  isGenerating,
  error,
  rateLimitRetryAfter,
}: CanvaGenerationPanelProps) {
  const [cooldownSeconds, setCooldownSeconds] = useState<number>(0);

  useEffect(() => {
    if (rateLimitRetryAfter && rateLimitRetryAfter > 0) {
      setCooldownSeconds(rateLimitRetryAfter);
    }
  }, [rateLimitRetryAfter]);

  useEffect(() => {
    if (cooldownSeconds <= 0) return;

    const timer = setInterval(() => {
      setCooldownSeconds((prev) => {
        if (prev <= 1) {
          clearInterval(timer);
          return 0;
        }
        return prev - 1;
      });
    }, 1000);

    return () => clearInterval(timer);
  }, [cooldownSeconds]);

  if (isGenerating) {
    return (
      <div className="flex items-center gap-3 rounded-lg border bg-white px-4 py-4">
        <SpinnerIcon />
        <div>
          <p className="text-sm font-semibold text-foreground">Dang tao Canva...</p>
          <p className="text-base text-muted-foreground">
            Qua trinh nay thuong mat 10–30 giay.
          </p>
        </div>
      </div>
    );
  }

  if (cooldownSeconds > 0) {
    const minutes = Math.ceil(cooldownSeconds / 60);
    return (
      <Alert className="border-destructive/50 bg-destructive/5">
        <AlertTitle className="text-destructive">Canva dang ban</AlertTitle>
        <AlertDescription className="text-destructive/80">
          Canva dang ban, vui long thu lai sau {minutes} phut.
        </AlertDescription>
      </Alert>
    );
  }

  if (error) {
    return (
      <Alert className="border-destructive/50 bg-destructive/5">
        <AlertTitle className="text-destructive">Khong the tao Canva</AlertTitle>
        <AlertDescription className="text-destructive/80">
          {error}
        </AlertDescription>
      </Alert>
    );
  }

  return null;
}
```

- Shows spinner with accent blue during generation
- Shows destructive alert for errors
- Shows rate-limit cooldown with countdown timer
- Returns null when idle (no unnecessary DOM)

"use client";

import { useState } from "react";
import { toast } from "sonner";
import { ExternalLink, Copy, RefreshCw } from "lucide-react";

import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { Button } from "@/components/ui/button";
import { Badge } from "@/components/ui/badge";

export interface CanvaArtifactState {
  status: string;
  designId: string | null;
  errorMessage: string | null;
  editUrl: string | null;
  viewUrl: string | null;
  thumbnailUrl: string | null;
}

interface CanvaResultCardProps {
  title: string;
  artifact: CanvaArtifactState;
  onRetry: () => void;
  isRetrying: boolean;
  isCooldown: boolean;
}

export function CanvaResultCard({
  title,
  artifact,
  onRetry,
  isRetrying,
  isCooldown,
}: CanvaResultCardProps) {
  const [copied, setCopied] = useState(false);

  const isSucceeded = artifact.status === "SUCCEEDED";
  const isFailed = artifact.status === "FAILED";
  const isProcessing = artifact.status === "PROCESSING";

  const handleCopyLink = async () => {
    if (!artifact.editUrl) return;
    try {
      await navigator.clipboard.writeText(artifact.editUrl);
      setCopied(true);
      toast.success("Da sao chep lien ket Canva.");
      setTimeout(() => setCopied(false), 2000);
    } catch {
      toast.error("Khong the sao chep lien ket.");
    }
  };

  return (
    <Card className={isFailed ? "border-destructive/30" : ""}>
      <CardHeader className="pb-3">
        <div className="flex items-center justify-between">
          <CardTitle className="text-sm font-semibold">{title}</CardTitle>
          {isSucceeded && (
            <Badge variant="outline" className="border-emerald-300 text-emerald-700">
              Thanh cong
            </Badge>
          )}
          {isFailed && (
            <Badge variant="outline" className="border-destructive text-destructive">
              That bai
            </Badge>
          )}
          {isProcessing && (
            <Badge variant="outline" className="border-[#3B82F6] text-[#3B82F6]">
              Dang tao...
            </Badge>
          )}
        </div>
      </CardHeader>
      <CardContent className="space-y-3">
        {/* Optional thumbnail */}
        {isSucceeded && artifact.thumbnailUrl && (
          <div className="overflow-hidden rounded-md border">
            <img
              src={artifact.thumbnailUrl}
              alt={`${title} preview`}
              className="h-auto w-full object-cover"
              loading="lazy"
            />
          </div>
        )}

        {/* Helper text */}
        {isSucceeded && (
          <p className="text-base text-muted-foreground">
            Lien ket Canva da san sang.
          </p>
        )}
        {isFailed && (
          <p className="text-base text-destructive/80">
            {artifact.errorMessage || "Khong the tao Canva luc nay. Hay thu lai."}
          </p>
        )}

        {/* Action row */}
        <div className="flex flex-wrap gap-2">
          {isSucceeded && artifact.editUrl && (
            <>
              <Button
                size="sm"
                className="gap-1.5 bg-[#3B82F6] text-white hover:bg-[#3B82F6]/90"
                asChild
              >
                <a href={artifact.editUrl} target="_blank" rel="noopener noreferrer">
                  <ExternalLink className="size-3.5" />
                  Mo trong Canva
                </a>
              </Button>
              <Button
                size="sm"
                variant="outline"
                className="gap-1.5"
                onClick={() => void handleCopyLink()}
              >
                <Copy className="size-3.5" />
                {copied ? "Da sao chep!" : "Sao chep link"}
              </Button>
            </>
          )}
          {isFailed && (
            <Button
              size="sm"
              variant="outline"
              className="gap-1.5 border-destructive text-destructive hover:bg-destructive/5"
              onClick={onRetry}
              disabled={isRetrying || isCooldown}

              {isRetrying ? "Dang thu lai..." : "Thu lai"}
            </Button>
          )}
        </div>
      </CardContent>
    </Card>
  );
}
