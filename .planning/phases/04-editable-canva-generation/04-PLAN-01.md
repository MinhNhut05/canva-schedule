# Plan 04-01: Externalized Template Mappings, Schema Extensions & Payload Builders

## Goal

Establish the data foundation for Canva generation: Prisma models for token persistence and per-artifact generation records, environment-driven template resolution for all 4 v1 templates, canonical field-name manifests, and typed payload builders that convert `StructuredDraft` into Canva autofill data. After this plan, the backend has everything it needs to construct Canva API requests without any hardcoded template IDs or magic field strings in business logic.

```typescript
import { getCanvaConfig } from "./server-client";

export type TemplateDuration = "ONE_DAY" | "TWO_DAY";
export type TemplateKind = "ITINERARY" | "MENU";
export type TemplateKey = `${TemplateDuration}_${TemplateKind}`;

export const ALL_TEMPLATE_KEYS: TemplateKey[] = [
  "ONE_DAY_ITINERARY",
  "ONE_DAY_MENU",
  "TWO_DAY_ITINERARY",
  "TWO_DAY_MENU",
];

/**
 * Display labels for template confirmation UI (Vietnamese).
 */
export const TEMPLATE_LABELS: Record<TemplateKey, string> = {
  ONE_DAY_ITINERARY: "Tour 1 ngay — Lich trinh",
  ONE_DAY_MENU: "Tour 1 ngay — Thuc don",
  TWO_DAY_ITINERARY: "Tour 2 ngay — Lich trinh",
  TWO_DAY_MENU: "Tour 2 ngay — Thuc don",
};

/**
 * Resolve the Canva Brand Template ID for a given duration and artifact kind.
 * Throws if the env var is missing or empty (caught by Zod in getCanvaEnv).
 */
export function resolveTemplateId(duration: TemplateDuration, kind: TemplateKind): string {
  const { templates } = getCanvaConfig();
  const key: TemplateKey = `${duration}_${kind}`;
  return templates[key];
}

/**
 * Resolve the template pair (itinerary + menu) for a tour duration.
 * Returns both template IDs and their display labels for the confirmation UI.
 */
export function resolveTemplatePair(duration: TemplateDuration) {
  return {
    itinerary: {
      templateId: resolveTemplateId(duration, "ITINERARY"),
      label: TEMPLATE_LABELS[`${duration}_ITINERARY`],
      kind: "ITINERARY" as TemplateKind,
    },
    menu: {
      templateId: resolveTemplateId(duration, "MENU"),
      label: TEMPLATE_LABELS[`${duration}_MENU`],
      kind: "MENU" as TemplateKind,
    },
  };
}
```

This module is the single source of truth for template ID lookup. No other module should read template env vars directly.

```typescript
    { prefix: "menu_day2", size: SLOT_SIZE },

/**
 * Generate sequential field names for a section: prefix_1, prefix_2, ..., prefix_N.
 */
export function generateSlotNames(prefix: string, size: number): string[] {
  return Array.from({ length: size }, (_, i) => `${prefix}_${i + 1}`);
}

/**
 * Pad an array to a fixed size, filling remaining slots with empty strings.
 */
export function toFixedTextSlots(values: string[], size: number): string[] {
  return Array.from({ length: size }, (_, i) => values[i] ?? "");
}
```

```typescript
import type { OneDayDraft, TwoDayDraft, StructuredDraft } from "@/lib/ai/extraction-schema";
import {
  SHARED_FIELDS,
  SLOT_SIZE,
  ONE_DAY_ITINERARY_FIELDS,
  ONE_DAY_MENU_FIELDS,
  TWO_DAY_ITINERARY_FIELDS,
  TWO_DAY_MENU_FIELDS,
  toFixedTextSlots,
  generateSlotNames,
} from "./field-map";

/** Canva autofill text field value shape. */
export interface CanvaTextField {
  type: "text";
  text: string;
}

/** Full Canva autofill data payload — record of field name to text value. */
export type CanvaAutofillData = Record<string, CanvaTextField>;

function textField(value: string): CanvaTextField {
  return { type: "text", text: value };
}

function buildSharedFields(draft: StructuredDraft): CanvaAutofillData {
  return {
    title: textField(draft.title ?? ""),
    client_name: textField(draft.clientName ?? ""),
    tour_date: textField(draft.tourDate ?? ""),
    greeting_text: textField(draft.greetingText ?? ""),
    pickup_location: textField(draft.pickupLocation ?? ""),
    return_location: textField(draft.returnLocation ?? ""),
  };
}

function buildSectionPayload(
  texts: string[],
  prefix: string,
  size: number,
): CanvaAutofillData {
  const slots = toFixedTextSlots(texts, size);
  const names = generateSlotNames(prefix, size);
  const result: CanvaAutofillData = {};
  for (let i = 0; i < size; i++) {
    result[names[i]] = textField(slots[i]);
  }
  return result;
}

export function buildOneDayItineraryPayload(draft: OneDayDraft): CanvaAutofillData {
  const shared = buildSharedFields(draft);
  const morning = draft.itinerary.morning.map((a) => a.text);
  const afternoon = draft.itinerary.afternoon.map((a) => a.text);

  return {
    ...shared,
    ...buildSectionPayload(morning, "morning", SLOT_SIZE),
    ...buildSectionPayload(afternoon, "afternoon", SLOT_SIZE),
  };
}

export function buildOneDayMenuPayload(draft: OneDayDraft): CanvaAutofillData {
  const shared = buildSharedFields(draft);
  const menuMorning = draft.menu.morning.map((m) => m.text);
  const menuLunch = draft.menu.lunch.map((m) => m.text);
  const menuAfternoon = draft.menu.afternoon.map((m) => m.text);

  return {
    ...shared,
    ...buildSectionPayload(menuMorning, "menu_morning", SLOT_SIZE),
    ...buildSectionPayload(menuLunch, "menu_lunch", SLOT_SIZE),
    ...buildSectionPayload(menuAfternoon, "menu_afternoon", SLOT_SIZE),
  };
}

export function buildTwoDayItineraryPayload(draft: TwoDayDraft): CanvaAutofillData {
  const shared = buildSharedFields(draft);
  const day1 = draft.itinerary.day1.map((a) => a.text);
  const day2 = draft.itinerary.day2.map((a) => a.text);

  return {
    ...shared,
    ...buildSectionPayload(day1, "day1", SLOT_SIZE),
    ...buildSectionPayload(day2, "day2", SLOT_SIZE),
  };
}

export function buildTwoDayMenuPayload(draft: TwoDayDraft): CanvaAutofillData {
  const shared = buildSharedFields(draft);
  const menuDay1 = draft.menu.day1.map((m) => m.text);
  const menuDay2 = draft.menu.day2.map((m) => m.text);

  return {
    ...shared,
    ...buildSectionPayload(menuDay1, "menu_day1", SLOT_SIZE),
    ...buildSectionPayload(menuDay2, "menu_day2", SLOT_SIZE),
  };
}

/**
 * Build the Canva autofill payload for a given artifact type from an approved draft.
 * This is the main entry point used by the generation server action.
 */
export function buildCanvaPayload(
  draft: StructuredDraft,
  artifactKind: "ITINERARY" | "MENU",
): CanvaAutofillData {
  if (draft.duration === "ONE_DAY") {
    return artifactKind === "ITINERARY"
      ? buildOneDayItineraryPayload(draft)
      : buildOneDayMenuPayload(draft);
  }
  return artifactKind === "ITINERARY"
    ? buildTwoDayItineraryPayload(draft)
    : buildTwoDayMenuPayload(draft);
}
```

This module converts the full `StructuredDraft` discriminated union into Canva-compatible `{ type: "text", text: "..." }` data payloads. Each builder handles one of the 4 template types. The `buildCanvaPayload` function is the unified entry point.
</action>
