```typescript
import "server-only";

import { prisma } from "@/lib/db";
import { getCanvaConfig } from "./server-client";

const CANVA_TOKEN_URL = "https://api.canva.com/rest/v1/oauth/token";

/** Buffer before actual expiry to trigger proactive refresh (5 minutes). */
const REFRESH_BUFFER_MS = 5 * 60 * 1000;

export interface CanvaTokenSet {
  accessToken: string;
  refreshToken: string;
  expiresAt: Date;
}

/**
 * Get the current valid access token.
 * If no token exists in DB, seed from env (first-time setup).
 * If token is expired or near-expiry, refresh it automatically.
 * Returns a valid accessToken string ready for API calls.
 */
export async function getValidAccessToken(): Promise<string> {
  let tokenRecord = await prisma.canvaToken.findFirst({
    orderBy: { updatedAt: "desc" },
  });

  // First-time seed from env if no DB token exists
  if (!tokenRecord) {
    tokenRecord = await seedTokenFromEnv();
  }

  const now = new Date();
  const expiresWithBuffer = new Date(tokenRecord.expiresAt.getTime() - REFRESH_BUFFER_MS);

  if (now >= expiresWithBuffer) {
    const refreshed = await refreshAccessToken(tokenRecord.refreshToken);
    await persistTokenSet(tokenRecord.id, refreshed);
    return refreshed.accessToken;
  }

  return tokenRecord.accessToken;
}

/**
 * Force-refresh the token (called on 401 retry).
 * Returns the new access token.
 */
export async function forceRefreshToken(): Promise<string> {
  const tokenRecord = await prisma.canvaToken.findFirst({
    orderBy: { updatedAt: "desc" },
  });

  if (!tokenRecord) {
    throw new Error("Khong tim thay token Canva. Hay cau hinh token ban dau.");
  }

  const refreshed = await refreshAccessToken(tokenRecord.refreshToken);
  await persistTokenSet(tokenRecord.id, refreshed);
  return refreshed.accessToken;
}

async function seedTokenFromEnv() {
  const initialAccessToken = process.env.CANVA_INITIAL_ACCESS_TOKEN;
  const initialRefreshToken = process.env.CANVA_INITIAL_REFRESH_TOKEN;

  if (!initialAccessToken || !initialRefreshToken) {
    throw new Error(
      "Khong tim thay token Canva trong DB va CANVA_INITIAL_ACCESS_TOKEN / CANVA_INITIAL_REFRESH_TOKEN chua duoc cau hinh."
    );
  }

  // Seed with 1-hour default expiry; will be refreshed on first use if expired
  return prisma.canvaToken.create({
    data: {
      accessToken: initialAccessToken,
      refreshToken: initialRefreshToken,
      expiresAt: new Date(Date.now() + 60 * 60 * 1000),
    },
  });
}

async function refreshAccessToken(refreshToken: string): Promise<CanvaTokenSet> {
  const { clientId, clientSecret } = getCanvaConfig();

  const params = new URLSearchParams({
    grant_type: "refresh_token",
    refresh_token: refreshToken,
  });

  const response = await fetch(CANVA_TOKEN_URL, {
    method: "POST",
    headers: {
      Authorization: `Basic ${Buffer.from(`${clientId}:${clientSecret}`).toString("base64")}`,
      "Content-Type": "application/x-www-form-urlencoded",
    },
    body: params.toString(),
    cache: "no-store",
  });

  if (!response.ok) {
    const body = await response.text().catch(() => "");
    throw new Error(`Canva token refresh that bai (${response.status}): ${body}`);
  }

  const data = await response.json();

  return {
    accessToken: data.access_token,
    refreshToken: data.refresh_token ?? refreshToken, // Keep old if Canva doesn't rotate
    expiresAt: new Date(Date.now() + (data.expires_in ?? 3600) * 1000),
  };
}

async function persistTokenSet(tokenId: string, tokenSet: CanvaTokenSet): Promise<void> {
  await prisma.canvaToken.update({
    where: { id: tokenId },
    data: {
      accessToken: tokenSet.accessToken,
      refreshToken: tokenSet.refreshToken,
      expiresAt: tokenSet.expiresAt,
    },
  });
}
```

Key behaviors:
- DB-first token lookup (survives restarts per D-25)
- First-time seed from `CANVA_INITIAL_ACCESS_TOKEN` / `CANVA_INITIAL_REFRESH_TOKEN` env vars
- Proactive refresh 5 min before expiry
- Force-refresh for 401 retry path
- Always persists rotated refresh token (Pitfall 5)

```typescript
import "server-only";

import { getValidAccessToken, forceRefreshToken } from "./oauth";

const CANVA_API_BASE = "https://api.canva.com/rest/v1";

export interface CanvaApiError {
  status: number;
  code: string;
  message: string;
  retryAfterSeconds?: number;
}

export class CanvaApiException extends Error {
  constructor(public readonly apiError: CanvaApiError) {
    super(apiError.message);
    this.name = "CanvaApiException";
  }

  get isRateLimited(): boolean {
    return this.apiError.status === 429;
  }

  get retryAfterSeconds(): number | undefined {
    return this.apiError.retryAfterSeconds;
  }
}

/**
 * Make an authenticated request to Canva REST API.
 * Automatically refreshes the token on 401 and retries once.
 * Throws CanvaApiException on non-2xx responses.
 */
export async function canvaFetch(
  path: string,
  options: RequestInit = {},
): Promise<Response> {
  const url = path.startsWith("http") ? path : `${CANVA_API_BASE}${path}`;

  const accessToken = await getValidAccessToken();

  const response = await doFetch(url, accessToken, options);

  // Retry once on 401 with a fresh token
  if (response.status === 401) {
    const freshToken = await forceRefreshToken();
    const retryResponse = await doFetch(url, freshToken, options);

    if (!retryResponse.ok) {
      throw await buildApiException(retryResponse);
    }

    return retryResponse;
  }

  if (!response.ok) {
    throw await buildApiException(response);
  }

  return response;
}

async function doFetch(
  url: string,
  accessToken: string,
  options: RequestInit,
): Promise<Response> {
  const headers: Record<string, string> = {
    Authorization: `Bearer ${accessToken}`,
    "Content-Type": "application/json",
    ...(options.headers as Record<string, string> | undefined),
  };

  return fetch(url, {
    ...options,
    headers,
    cache: "no-store",
  });
}

async function buildApiException(response: Response): Promise<CanvaApiException> {
  let code = "UNKNOWN_ERROR";
  let message = `Canva API loi (${response.status})`;

  try {
    const body = await response.json();
    code = body.code ?? body.error ?? code;
    message = body.message ?? body.error_description ?? message;
  } catch {
    // Fallback to status-based message
  }

  const retryAfterHeader = response.headers.get("Retry-After");
  const retryAfterSeconds = retryAfterHeader ? parseInt(retryAfterHeader, 10) : undefined;

  return new CanvaApiException({
    status: response.status,
    code,
    message,
    retryAfterSeconds: Number.isFinite(retryAfterSeconds) ? retryAfterSeconds : undefined,
  });
}
```

Key behaviors:
- All Canva API calls go through `canvaFetch`
- Auto-refresh on 401 + single retry
- Rate limit detection with Retry-After header parsing
- CanvaApiException with typed error for downstream handling

```typescript
import "server-only";

import { canvaFetch, CanvaApiException } from "./client";

/** Polling delays in milliseconds: 2s, 3s, 5s, 5s, 5s, 5s (total ~25s max wait). */
const POLL_DELAYS_MS = [2000, 3000, 5000, 5000, 5000, 5000];

export interface AutofillJobResult {
  status: "success" | "failed";
  designId?: string;
  editUrl?: string;
  viewUrl?: string;
  thumbnailUrl?: string;
  errorCode?: string;
  errorMessage?: string;
}

/**
 * Create a Canva design autofill job.
 * Uses Brand Template ID + data payload.
 * Returns the job ID for polling.
 */
export async function createAutofillJob(
  brandTemplateId: string,
  title: string,
  data: Record<string, unknown>,
): Promise<string> {
  const response = await canvaFetch("/autofills", {
    method: "POST",
    body: JSON.stringify({
      brand_template_id: brandTemplateId,
      title,
      data,
    }),
  });

  const payload = await response.json();
  const jobId = payload?.job?.id;

  if (!jobId) {
    throw new Error("Canva API khong tra ve job ID.");
  }

  return jobId as string;
}

/**
 * Poll a Canva autofill job until it reaches a terminal state (success or failed).
 * Uses bounded polling with increasing delays to avoid self-inflicted rate limits.
 * Throws on timeout or failed job.
 */
export async function pollAutofillJob(jobId: string): Promise<AutofillJobResult> {
  for (const delayMs of POLL_DELAYS_MS) {
    await sleep(delayMs);

    let response: Response;
    try {
      response = await canvaFetch(`/autofills/${jobId}`);
    } catch (error) {
      if (error instanceof CanvaApiException && error.isRateLimited) {
        // Wait longer on rate limit, then continue polling
        await sleep(error.retryAfterSeconds ? error.retryAfterSeconds * 1000 : 10000);
        continue;
      }
      throw error;
    }

    const payload = await response.json();
    const job = payload?.job;

    if (!job) {
      throw new Error("Canva API tra ve response khong hop le khi poll job.");
    }

    if (job.status === "success") {
      const design = job.result?.design;
      return {
        status: "success",
        designId: design?.id,
        editUrl: design?.urls?.edit_url,
        viewUrl: design?.urls?.view_url,
        thumbnailUrl: design?.thumbnail?.url,
      };
    }

    if (job.status === "failed") {
      return {
        status: "failed",
        errorCode: job.error?.code,
        errorMessage: job.error?.message ?? "Canva autofill job that bai.",
      };
    }

    // status is "in_progress" — continue polling
  }

  // Exhausted all polling attempts
  return {
    status: "failed",
    errorCode: "TIMEOUT",
    errorMessage: "Canva generation het thoi gian cho. Vui long thu lai.",
  };
}

/**
 * Create an autofill job and poll it to completion.
 * Returns the final result with design metadata or error info.
 */
export async function executeAutofillJob(
  brandTemplateId: string,
  title: string,
  data: Record<string, unknown>,
): Promise<AutofillJobResult> {
  const jobId = await createAutofillJob(brandTemplateId, title, data);
  return pollAutofillJob(jobId);
}

function sleep(ms: number): Promise<void> {
  return new Promise((resolve) => setTimeout(resolve, ms));
}
```

Key behaviors:
- Bounded polling with delays [2s, 3s, 5s, 5s, 5s, 5s] (~25s max)
- Rate limit awareness during polling (waits Retry-After)
- Timeout returns as "failed" result, not thrown exception
- `executeAutofillJob` is the primary entry point

```typescript
import "server-only";

import { prisma } from "@/lib/db";
import type { StructuredDraft } from "@/lib/ai/extraction-schema";
import { resolveTemplateId } from "./template-resolver";
import type { TemplateDuration, TemplateKind } from "./template-resolver";
import { buildCanvaPayload } from "./payload";
import { executeAutofillJob, type AutofillJobResult } from "./jobs";
import { getFreshDesignLinks, type DesignLinks } from "./designs";
import { CanvaApiException } from "./client";
import { CANVA_ARTIFACT_STATUS, ARTIFACT_TYPE } from "./canva-status";

export interface GenerateArtifactResult {
  artifactType: "ITINERARY" | "MENU";
  status: "SUCCEEDED" | "FAILED";
  designId?: string;
  editUrl?: string;
  viewUrl?: string;
  thumbnailUrl?: string;
  errorCode?: string;
  errorMessage?: string;
  rateLimitRetryAfter?: number;
}

export interface GeneratePairResult {
  itinerary: GenerateArtifactResult;
  menu: GenerateArtifactResult;
}

/**
 * Generate both Canva artifacts (Itinerary + Menu) for an approved draft.
 * Runs both jobs in parallel. Persists results independently (partial success OK).
 * On re-generation, upserts existing artifact records.
 */
export async function generateCanvaPair(
  uploadId: string,
  draft: StructuredDraft,
): Promise<GeneratePairResult> {
  const duration: TemplateDuration = draft.duration === "ONE_DAY" ? "ONE_DAY" : "TWO_DAY";
  const title = draft.title ?? "SileTravel — generated";

  // Mark both artifacts as PROCESSING
  await upsertArtifact(uploadId, "ITINERARY", {
    status: CANVA_ARTIFACT_STATUS.PROCESSING,
    templateId: resolveTemplateId(duration, "ITINERARY"),
    designId: null,
    jobId: null,
    errorMessage: null,
  });
  await upsertArtifact(uploadId, "MENU", {
    status: CANVA_ARTIFACT_STATUS.PROCESSING,
    templateId: resolveTemplateId(duration, "MENU"),
    designId: null,
    jobId: null,
    errorMessage: null,
  });

  // Build payloads
  const itineraryPayload = buildCanvaPayload(draft, "ITINERARY");
  const menuPayload = buildCanvaPayload(draft, "MENU");

  // Execute both jobs in parallel
  const [itineraryOutcome, menuOutcome] = await Promise.allSettled([
    executeOneArtifact(uploadId, duration, "ITINERARY", `${title} — Lich trinh`, itineraryPayload),
    executeOneArtifact(uploadId, duration, "MENU", `${title} — Thuc don`, menuPayload),
  ]);

  const itinerary = resolveOutcome("ITINERARY", itineraryOutcome);
  const menu = resolveOutcome("MENU", menuOutcome);

  return { itinerary, menu };
}

async function executeOneArtifact(
  uploadId: string,
  duration: TemplateDuration,
  kind: TemplateKind,
  title: string,
  data: Record<string, unknown>,
): Promise<GenerateArtifactResult> {
  const templateId = resolveTemplateId(duration, kind);

  try {
    const result: AutofillJobResult = await executeAutofillJob(templateId, title, data);

    if (result.status === "success" && result.designId) {
      await upsertArtifact(uploadId, kind, {
        status: CANVA_ARTIFACT_STATUS.SUCCEEDED,
        designId: result.designId,
        jobId: null,
        errorMessage: null,
      });

      return {
        artifactType: kind,
        status: "SUCCEEDED",
        designId: result.designId,
        editUrl: result.editUrl,
        viewUrl: result.viewUrl,
        thumbnailUrl: result.thumbnailUrl,
      };
    }

    // Job failed
    await upsertArtifact(uploadId, kind, {
      status: CANVA_ARTIFACT_STATUS.FAILED,
      errorMessage: result.errorMessage ?? "Canva autofill that bai.",
    });

    return {
      artifactType: kind,
      status: "FAILED",
      errorCode: result.errorCode,
      errorMessage: result.errorMessage,
    };
  } catch (error) {
    const isRateLimited = error instanceof CanvaApiException && error.isRateLimited;
    const errorMessage =
      error instanceof Error ? error.message : "Loi khong xac dinh khi tao Canva.";
    const retryAfter =
      error instanceof CanvaApiException ? error.retryAfterSeconds : undefined;

    await upsertArtifact(uploadId, kind, {
      status: CANVA_ARTIFACT_STATUS.FAILED,
      errorMessage,
    });

    return {
      artifactType: kind,
      status: "FAILED",
      errorCode: isRateLimited ? "RATE_LIMITED" : "API_ERROR",
      errorMessage,
      rateLimitRetryAfter: retryAfter,
    };
  }
}

function resolveOutcome(
  kind: "ITINERARY" | "MENU",
  outcome: PromiseSettledResult<GenerateArtifactResult>,
): GenerateArtifactResult {
  if (outcome.status === "fulfilled") {
    return outcome.value;
  }

  return {
    artifactType: kind,
    status: "FAILED",
    errorCode: "UNEXPECTED_ERROR",
    errorMessage:
      outcome.reason instanceof Error
        ? outcome.reason.message
        : "Loi khong xac dinh.",
  };
}

async function upsertArtifact(
  uploadId: string,
  artifactType: string,
  data: {
    status: string;
    templateId?: string | null;
    designId?: string | null;
    jobId?: string | null;
    errorMessage?: string | null;
  },
): Promise<void> {
  await prisma.canvaArtifact.upsert({
    where: {
      uploadId_artifactType: { uploadId, artifactType },
    },
    update: {
      ...data,
    },
    create: {
      uploadId,
      artifactType,
      ...data,
    },
  });

/**
 * Resolve fresh Canva edit/view URLs for all persisted artifacts of an upload.
 * Called on review page load to get non-expired URLs.
 * Returns null for artifacts that don't exist or don't have a designId.
 */
export async function resolveArtifactLinks(
  uploadId: string,
): Promise<{ itinerary: DesignLinks | null; menu: DesignLinks | null }> {
  const artifacts = await prisma.canvaArtifact.findMany({
    where: { uploadId },
  });

  const itineraryArtifact = artifacts.find((a) => a.artifactType === ARTIFACT_TYPE.ITINERARY);
  const menuArtifact = artifacts.find((a) => a.artifactType === ARTIFACT_TYPE.MENU);

  const [itineraryLinks, menuLinks] = await Promise.allSettled([
    itineraryArtifact?.designId
      ? getFreshDesignLinks(itineraryArtifact.designId)
      : Promise.resolve(null),
    menuArtifact?.designId
      ? getFreshDesignLinks(menuArtifact.designId)
      : Promise.resolve(null),
  ]);

  return {
    itinerary: itineraryLinks.status === "fulfilled" ? itineraryLinks.value : null,
    menu: menuLinks.status === "fulfilled" ? menuLinks.value : null,
  };
}

/**
 * Get persisted artifact records for an upload (status, errors, designIds).
 * Used by the review page to determine UI state without calling Canva API.
 */
export async function getArtifactStates(uploadId: string) {
  const artifacts = await prisma.canvaArtifact.findMany({
    where: { uploadId },
  });

  return {
    itinerary: artifacts.find((a) => a.artifactType === ARTIFACT_TYPE.ITINERARY) ?? null,
    menu: artifacts.find((a) => a.artifactType === ARTIFACT_TYPE.MENU) ?? null,
  };
}

/**
 * Retry a single failed artifact.
 * Re-builds payload from the current approved draft and re-executes just that artifact.
 */
export async function retryArtifact(
  uploadId: string,
  kind: "ITINERARY" | "MENU",
  draft: StructuredDraft,
): Promise<GenerateArtifactResult> {
  const duration: TemplateDuration = draft.duration === "ONE_DAY" ? "ONE_DAY" : "TWO_DAY";
  const title = draft.title ?? "SileTravel — generated";
  const payload = buildCanvaPayload(draft, kind);

  await upsertArtifact(uploadId, kind, {
    status: CANVA_ARTIFACT_STATUS.PROCESSING,
    designId: null,
    jobId: null,
    errorMessage: null,
  });

  return executeOneArtifact(
    uploadId,
    duration,
    kind,
    `${title} — ${kind === "ITINERARY" ? "Lich trinh" : "Thuc don"}`,
    payload,
  );
}
```
