# Plan 06-02: Pipeline Reliability Hardening

## Goal

Make the backend pipeline failure-tolerant and volume-safe for ~10 tours/week: global Canva rate-limit cooldown persisted in DB so all users see it, AI extraction timeout at 30s with exponential backoff, Canva polling timeout at 2 minutes, and initial cooldown state passed from SSR to the review UI.

## must_haves (goal-backward verification)

- [ ] CanvaToken model has nullable `cooldownUntil` DateTime field
- [ ] `getGlobalCooldown()` and `setGlobalCooldown()` helpers exist in `src/lib/canva/cooldown.ts`
- [ ] AI extraction uses exponential backoff (1s, 2s, 4s) instead of linear (1s, 2s)
- [ ] AI extraction aborts after 30 seconds with a Vietnamese error message
- [ ] Canva polling aborts after 2 minutes wall-clock time
- [ ] `generateCanva()` and `retryCanvaArtifact()` check global cooldown before starting
- [ ] `generateArtifact()` writes cooldown to DB when Canva rate limit is hit
- [ ] Review page SSR queries `cooldownUntil` and passes it as prop to `ReviewPage`
- [ ] `ReviewPage` accepts `initialCooldown` prop and initializes cooldown state from it

import "server-only";
import { db } from "@/lib/db";

export async function getGlobalCooldown(): Promise<Date | null> {
  const token = await db.canvaToken.findFirst({
    orderBy: { updatedAt: "desc" },
    select: { cooldownUntil: true },
  });
  if (!token?.cooldownUntil) return null;
  if (token.cooldownUntil.getTime() <= Date.now()) return null;
  return token.cooldownUntil;
}

export async function setGlobalCooldown(cooldownSeconds: number): Promise<void> {
  const cooldownUntil = new Date(Date.now() + cooldownSeconds * 1000);
  const token = await db.canvaToken.findFirst({
    orderBy: { updatedAt: "desc" },
    select: { id: true },
  });
  if (!token) return;
  await db.canvaToken.update({
    where: { id: token.id },
    data: { cooldownUntil },
  });
}

export function getRemainingCooldownSeconds(cooldownUntil: Date): number {
  const remaining = Math.ceil((cooldownUntil.getTime() - Date.now()) / 1000);
  return Math.max(0, remaining);
}
```

export async function pollAutofillJob(jobId: string): Promise<AutofillJobResult> {
  const deadline = Date.now() + POLL_TIMEOUT_MS;

  for (const delayMs of POLL_DELAYS_MS) {
    await sleep(delayMs);

    if (Date.now() > deadline) {
      throw new Error("Canva tao thiet ke qua lau (vuot 2 phut). Vui long thu lai.");
    }

    const response = await canvaFetch(`/autofills/${jobId}`);
    if (!response.ok) {
      throw new Error(`Canva poll failed (${response.status})`);
    }

    const payload = await response.json();
    const job = payload?.job;

    if (job?.status === "success") {
      const design = job?.result?.design;
      return {
        designId: design?.id ?? "",
        editUrl: design?.urls?.edit_url ?? "",
        viewUrl: design?.urls?.view_url ?? "",
        thumbnailUrl: design?.thumbnail?.url,
      };
    }

    if (job?.status === "failed") {
      throw new Error(job?.error?.message ?? "Canva autofill failed");
    }
  }

  throw new Error("Canva generation timed out after polling");
}
```

useEffect(() => {

    const remaining = cooldownUntil!.getTime() - Date.now();
    if (remaining <= 0) {
      setIsRateLimited(false);
      setCooldownUntil(null);
      setCooldownMinutes(0);
      return;
    }
    setCooldownMinutes(Math.max(1, Math.ceil(remaining / 60000)));
