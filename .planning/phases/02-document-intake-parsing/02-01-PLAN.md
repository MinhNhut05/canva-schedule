# Plan 02-01: Foundation — shadcn/ui, Types, Fixtures, Prisma Schema

## Goal

Set up all infrastructure for Phase 2: initialize shadcn/ui with Tailwind CSS, define shared TypeScript types and constants, create test fixtures for PDF/DOCX testing, configure Vitest, and add the Upload model to the database. This is Wave 0 — everything else in Phase 2 depends on these artifacts.

  <action>
    Create `src/lib/documents/types.ts` with these exact type definitions:

    ```typescript
    export type DocumentKind = "pdf" | "docx";

    export type UploadStatus =
      | "PENDING"
      | "PROCESSING"
      | "COMPLETED"
      | "COMPLETED_WITH_WARNING"
      | "FAILED";

    /** Maps to UI badges: "Tốt", "Cần kiểm tra", "Thấp" */
    export type QualityLevel = "good" | "warning" | "poor";

    export type QualityFlag =
      | "textTooShort"
      | "containsReplacementChars"
      | "lowVietnameseSignal"
      | "highSymbolNoise"
      | "likelyScannedPdf"
      | "fragmentedDocx";

    export interface QualityResult {
      score: number;
      level: QualityLevel;
      flags: QualityFlag[];
    }

    export interface ExtractionResult {
      kind: DocumentKind;
      originalFileName: string;
      mime: string;
      sizeBytes: number;
      rawText: string;
      normalizedText: string;
      pageCount?: number;
      warnings: string[];
      quality: QualityResult;
      processingTimeMs: number;
    }

    export interface FileValidationResult {
      valid: boolean;
      error?: string;
      kind?: DocumentKind;
      detectedMime?: string;
    }

    export interface UploadApiResponse {
      success: boolean;
      data?: ExtractionResult & { uploadId: string };
      error?: string;
    }

    export const ALLOWED_MIMES: Record<string, DocumentKind> = {
      "application/pdf": "pdf",
      "application/vnd.openxmlformats-officedocument.wordprocessingml.document": "docx",
    };

    export const ALLOWED_EXTENSIONS = [".pdf", ".docx"] as const;

    export const MAX_FILE_SIZE_BYTES = 30 * 1024 * 1024;

    export const MAX_FILE_SIZE_LABEL = "30MB";
    ```

    2. Add Upload model:
       ```prisma
       model Upload {
         id               String       @id @default(cuid())
         userId           String
         originalFileName String
         detectedMime     String?
         sourceKind       String?
         sizeBytes        Int
         status           UploadStatus @default(PENDING)
         rawText          String?      @db.Text
         normalizedText   String?      @db.Text
         qualityScore     Float?
         qualityLevel     String?
         qualityFlags     String[]     @default([])
         warningMessages  String[]     @default([])
         errorMessage     String?
         processingTimeMs Int?
         createdAt        DateTime     @default(now())
         updatedAt        DateTime     @updatedAt

         user User @relation(fields: [userId], references: [id])

         @@map("uploads")
       }
       ```

    3. Add `uploads Upload[]` relation to existing User model.

    4. Run migration:
       ```bash
       npx prisma migrate dev --name add-upload-model
       npx prisma generate
       ```
